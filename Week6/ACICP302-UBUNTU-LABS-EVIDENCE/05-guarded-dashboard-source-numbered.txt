   223	app = Flask(__name__)
   224	state = {"temp":0,"pressure":0,"flow":0,"alarm":0,"connected":False}
   225	state_lock = threading.Lock()
   226	
   227	def poll_modbus():
   228	    client = ModbusTcpClient(TARGET_HOST, port=TARGET_PORT)
   229	    while True:
   230	        try:
   231	            if not client.connect():
   232	                with state_lock: state["connected"] = False
   233	                time.sleep(1); continue
   234	            r = client.read_holding_registers(0,4,slave=1)
   235	            if not r.isError():
   236	                with state_lock:
   237	                    state["temp"]=r.registers[0]
   238	                    state["pressure"]=r.registers[1]
   239	                    state["flow"]=r.registers[2]
   240	                    state["alarm"]=r.registers[3]
   241	                    state["connected"]=True
   242	        except:
   243	            with state_lock: state["connected"] = False
   244	        time.sleep(1)
   245	
   246	def read_events(n=50):
   247	    if not os.path.exists(EVENT_LOG_FILE): return []
   248	    try:
   249	        lines = open(EVENT_LOG_FILE).readlines()[-n:]
   250	        return [json.loads(l) for l in lines if l.strip()]
   251	    except: return []
   252	
   253	@app.route("/api/state")
   254	def api_state():
   255	    with state_lock: return jsonify(dict(state))
   256	
   257	@app.route("/api/events")
   258	def api_events(): return jsonify(read_events())
   259	
   260	@app.route("/")
   261	def index(): return render_template_string(PAGE)
   262	
   263	if __name__ == "__main__":
   264	    threading.Thread(target=poll_modbus,daemon=True).start()
   265	    app.run(host="127.0.0.1",port=8081)
